# GWR Multiplayer Pack on_actions.txt

on_actions = {
    on_startup = {
        effect = { 
            GWR = { on_startup_handle_global = yes }
            every_country = { on_startup_handle_country = yes } 
            every_state = { on_startup_handle_state = yes }
        }
    }

    on_monthly = {
        effect = {
            GWR = { on_monthly_handle_global = yes }
            every_country = { on_monthly_handle_country = yes } 
        }
    }
    
    on_leave_faction = {
        effect = {
            ROOT = {
                on_leave_faction_handle_country = yes 
            }
        }
    }

    on_join_faction = {
        effect = {
            FROM = {
                add_to_faction = ROOT 
                on_join_faction_handle_leader = yes 
            }
        }
    }
}
