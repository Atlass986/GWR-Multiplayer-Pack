				clr_country_flag = warning_faction_sender
				clr_country_flag = warning_sender
			}
		}

		complete_effect = {
			THIS = {
				every_country = {
					limit = { is_in_faction_with = PREV }
					add_timed_idea = { idea = warning_30d days = 35 } 
					set_country_flag = warning_receiver
					set_variable = { war_with = ROOT }

				set_variable = { war_with = PREV.THIS }
				set_variable = { leader_war_with = ROOT }
