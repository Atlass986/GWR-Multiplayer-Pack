set_country_cost = {
	if = {
        limit = { is_ai = yes  }
        set_variable = { country_cost = 0 }
    } 
    else_if = {
        limit = { is_super_major = yes  }
        set_variable = { country_cost = 4 }
    } 
    else_if = {
        limit = { is_major_country = yes  }
        set_variable = { country_cost = 3 }
    } 
    else_if = {
        limit = { is_regional = yes  }
        set_variable = { country_cost = 2 }
    } 
    else_if = {
        limit = { is_minor = yes }
        set_variable = { country_cost = 1 }
    } 
    else_if = {
        limit = { is_mini = yes  }
        set_variable = { country_cost = 0.5 }
    } 
    else = {
        set_variable = { country_cost = 0 }
    }
}

set_faction_points = {
    set_variable = { faction_points = country_cost }
    every_other_country = {
        limit = { is_in_faction_with = PREV is_faction_leader = no is_ai = no }
        set_variable = { faction_points = 0 }
        set_country_cost = yes
        PREV = { add_to_variable = { faction_points = PREV.country_cost } }
        add_to_temp_array = { 
            array = temp_countries
            value = THIS
        }
    }
    for_each_scope_loop = {
        array = temp_countries
        set_variable = { faction_points = PREV.faction_points } 
    }
    clear_temp_array = temp_countries
}

	}
	else_if = {
		limit = { is_in_faction = no }
		set_country_cost = yes 
		set_variable = { faction_points = 0 }
	}
}
