scripted_gui = {
	
	state_interactions_menu_gui = {
		context_type = selected_state_context
		window_name = "state_interaction_menu_container"
		parent_window_name = "countrystateview"
		visible = { ROOT = { is_ai = no } }
		properties = {
			interaction_debug_button = {
				frame = global.SI_debug_var
			}
		}
		effects = {
			state_interaction_menu_button_click = {
				if = {
					limit = {
						ROOT = { 
							NOT = { has_country_flag = SI_menu_2_flag } 
						}
					}
					ROOT = { set_country_flag = SI_menu_2_flag}
				}
				else = {
					ROOT = { clr_country_flag = SI_menu_2_flag }
				}
			}
			interaction_debug_button_click = {
				if = {
					limit = { check_variable = { global.SI_debug_var = 1 } }
					set_variable = { global.SI_debug_var = 2 }
					set_global_flag = SI_debug_flag
				}
				else = {
					set_variable = { global.SI_debug_var = 1 }
					clr_global_flag = SI_debug_flag
				}
			}

		}
		triggers = {
			interaction_debug_button_visible = { is_debug = yes }
		}
	}

	state_interactions_transfer_tool_gui = {

		context_type = selected_state_context

		window_name = "state_interactions_state_transfer_container"

		parent_window_name = "countrystateview"

		visible = {
			ROOT = { has_country_flag = SI_menu_2_flag is_ai = no }
		}

		effects = {
			send_message_diplo_action_button_click = {
				ROOT = {
					set_variable = { selected_country_to_message = PREV.PREV.OWNER }
				}
			}
			send_message_diplo_action_selected_button_click = {
			}

			state_menu_transfer_menu_button_click = {
				if = {
					limit = { ROOT = { has_country_flag = SI_menu_2_flag } }
					ROOT = { clr_country_flag = SI_menu_2_flag }
				}
				else = {
					ROOT = { set_country_flag = SI_menu_2_flag }
				}
			}

			state_menu_demand_state_button_click = {
				set_variable = { ROOT.requested_state_@FROM.owner = FROM }
				FROM.owner = {
					country_event = {
						id = sievent.2
						hours = 1
					}
				}
				FROM = {
					set_state_flag = {
						flag = SI_state_demand_subject_flag
						days = 21
					}
				}
				ROOT = {
					add_political_power = var:global.si_demand_state_pp_take
					add_timed_idea = {
						idea = SI_major_demands_idea
						days = var:global.si_demand_state_cooldown_days
					}
				}
				# Debug
				if = {
					limit = { is_debug = yes }
					log = "(SI) [ROOT.GetName] demanded [FROM.GetName] from [FROM.owner.GetName]"
				}
			}

			state_menu_demand_vsl_button_click = {
				FROM.OWNER = {
					country_event = {
						id = sievent.4
						hours = 1
					}
				}
				
				ROOT = {
					add_political_power = var:global.si_demand_vassalisation_pp_take
					add_timed_idea = {
						idea = SI_major_demands_idea
						days = var:global.si_demand_vassalisation_cooldown_days
					}
					set_country_flag = {
						flag = SI_country_demand_subject_flag
						days = 21
					}
				}
				# Debug
				if = {
					limit = { is_debug = yes }
					log = "(SI) [ROOT.GetName] demanded [FROM.owner.GetName] vassalisation"
				}
			}

			state_menu_claim_state_button_click = {
				if = {
					limit = {
						NOT = { has_global_flag = SI_debug_flag }
					}
					ROOT = {
						add_command_power = var:global.si_claim_state_cp_take
						add_timed_idea = {
							idea = SI_minor_demands_idea
							days = var:global.si_claim_state_cooldown_days
						}
					}
				}
				FROM = { add_claim_by = ROOT }
			}

			state_menu_move_capital_button_click = {
				if = {
					limit = {
						NOT = { has_global_flag = SI_debug_flag }
					}
					ROOT = {
						add_political_power = var:global.si_move_capital_pp_take
						add_timed_idea = {
							idea = SI_else_demands_idea
							days = var:global.si_move_capital_cooldown_days
						}
					}
				}
				ROOT = { set_capital = { state = PREV } }
			}

			state_menu_ask_resources_button_click = {
				set_variable = { ROOT.requested_resource_state = FROM }
				if = {
					limit = {
						NOT = { has_global_flag = SI_debug_flag }
					}
					ROOT = {
						add_political_power = var:global.si_ask_resources_pp_take
						add_timed_idea = {
							idea = SI_else_demands_idea
							days = var:global.si_ask_resources_cooldown_days
						}
					}
				}
				FROM.OWNER = {
					country_event = { id = sievent.8 hours = 1 }
				}
			}

			state_transfer_mark_button_click = {
				add_to_array = {
					array = ROOT.selected_states_array
					value = FROM.id
				}
			}
			state_transfer_unmark_button_click = {
				remove_from_array = {
					array = ROOT.selected_states_array
					value = FROM.id
				}
			}

			state_transfer_unmark_all_button_click = {
				ROOT = { clear_array = THIS.selected_states_array }
			}

			state_transfer_transfer_button_click = {
				for_each_scope_loop = {
					array = ROOT.selected_states_array
					if = { 
						limit = { is_owned_and_controlled_by = ROOT }
						#ROOT = { add_political_power = var:global.si_demand_state_pp_take  } 
						transfer_state_to = FROM.owner
					}
				}
				ROOT = { clear_array = THIS.selected_states_array }
			}
			gwrmp_si_select_resources_button_click = {
				if = {
					limit = { ROOT = { has_country_flag = gwrmp_si_select_resources_visible } }
					ROOT = { clr_country_flag = gwrmp_si_select_resources_visible }
				}
				else = {
					ROOT = { set_country_flag = gwrmp_si_select_resources_visible }
				}
			}
		}

		triggers = {
			sr_resources_entry_container_visible = {
				has_country_flag = gwrmp_si_select_resources_visible
			}
			send_message_diplo_action_button_visible = {
				ROOT = {
					NOT = { check_variable = { selected_country_to_message = PREV.PREV.OWNER } }
				}
			}
			send_message_diplo_action_button_click_enabled = {
				FROM.OWNER = { NOT = { tag = ROOT } }
			}
			send_message_diplo_action_selected_button_visible = {
				ROOT = {
					check_variable = { selected_country_to_message = PREV.PREV.OWNER }
				}
			}
			state_menu_transfer_menu_button_visible = {	}
			send_message_diplo_action_button_click_enabled = {
				FROM.OWNER = {
					NOT = { tag = ROOT is_ai = yes }
				}
				ROOT = {
					NOT = { check_variable = { selected_country_to_message = PREV.PREV.OWNER } }
				}
			}

			state_menu_demand_state_button_click_enabled = {
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						ROOT = {
							has_political_power > var:global.si_demand_state_pp_required
							is_subject = no
							NOT = { has_idea = SI_major_demands_idea }
						}
						FROM.owner = {
							NOT = {
								is_ai = yes
								tag = ROOT
								tag = GWR
								has_war_with = ROOT
								is_subject_of = ROOT
								is_in_faction_with = ROOT
							}
							OR = {
								is_neighbor_of = ROOT
								any_neighbor_country = { is_subject_of = ROOT }
							}
						}
						FROM = {
							any_neighbor_state = {
								owner = {
									OR = {
										is_subject_of = ROOT
										tag = ROOT
									}
								}
							}
							NOT = { has_state_flag = SI_state_demand_subject_flag }
						}
					}
				}
			}
			state_menu_demand_vsl_button_click_enabled = {
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						ROOT = {
							has_political_power > var:global.si_demand_vassalisation_pp_required
							is_subject = no
							NOT = { has_idea = SI_major_demands_idea }
						}
						FROM.owner = {
							is_major = no
							is_ai = no
							has_war = no
							NOT = {
								tag = ROOT
								tag = GWR
								is_subject_of = ROOT
								is_in_faction_with = ROOT
								has_country_flag = SI_country_demand_subject_flag
							}
							OR = {
								is_neighbor_of = ROOT
								any_neighbor_country = { is_subject_of = ROOT }
							}
						}
					}
				}
			}
			state_menu_claim_state_button_click_enabled = {
				ROOT = { is_ai = no }
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						ROOT = {
							command_power > var:global.si_claim_state_cp_required
							NOT = { has_idea = SI_minor_demands_idea }
						}
						OR = {
							AND = {
								FROM.owner = {
									NOT = {
										is_ai = yes
										tag = ROOT
										tag = GWR
										is_in_faction_with = ROOT
										is_subject_of = ROOT
									}
									OR = {
										is_neighbor_of = ROOT
										any_neighbor_country = { is_subject_of = ROOT }
									}
								}
								FROM = {
									NOT = {
										is_core_of = ROOT
										is_claimed_by = ROOT
									}
								}
							}
						}
					}
				}
			}

			state_menu_move_capital_button_click_enabled = {
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						FROM = {
							is_owned_by = ROOT
							is_capital = no
							is_core_of = ROOT
						}
						ROOT = {
							has_political_power > var:global.si_move_capital_pp_required
							NOT = { has_idea = SI_else_demands_idea }
						}
					}
				}
			}

			state_menu_ask_resources_button_click_enabled = {
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						FROM = { 
							NOT = { is_owned_by = ROOT } 
							OR = {
								AND = { 
									oil > 0 
									ROOT = { has_country_flag = GWRMP_SI_OIL } 
									NOT = { FROM.OWNER = { any_other_country = { has_resources_rights = { state = FROM resources = { oil } } } } }
								}
								AND = { 
									aluminium > 0 
									ROOT = { has_country_flag = GWRMP_SI_ALUMINUM } 
									NOT = { FROM.OWNER = { any_other_country = { has_resources_rights = { state = FROM resources = { aluminium } } } } } 
								}
								AND = { 
									rubber > 0 
									ROOT = { has_country_flag = GWRMP_SI_RUBBER } 
									NOT = { FROM.OWNER = { any_other_country = { has_resources_rights = { state = FROM resources = { rubber } } } } } 
								}
								AND = { 
									tungsten > 0 
									ROOT = { has_country_flag = GWRMP_SI_TUNGSTEN } 
									NOT = { FROM.OWNER = { any_other_country = { has_resources_rights = { state = FROM resources = { tungsten } } } } } 
								}
								AND = { 
									steel > 0 
									ROOT = { has_country_flag = GWRMP_SI_STEEL } 
									NOT = { FROM.OWNER = { any_other_country = { has_resources_rights = { state = FROM resources = { steel } } } } } 
								}
								AND = { 
									chromium > 0 
									ROOT = { has_country_flag = GWRMP_SI_CHROMIUM } 
									NOT = { FROM.OWNER = { any_other_country = { has_resources_rights = { state = FROM resources = { chromium } } } } } 
								}
								AND = { 
									coal > 0 
									ROOT = { has_country_flag = GWRMP_SI_COAL } 
									NOT = { FROM.OWNER = { any_other_country = { has_resources_rights = { state = FROM resources = { coal } } } } } 
								}
							}
						} 
						FROM.OWNER = { 
							NOT = { 
								is_ai = yes
								tag = GWR
								has_war_with = ROOT 
								any_other_country = { has_resources_rights = { state = PREV.PREV } }
							} 
						}
						ROOT = {
							OR = { 
								has_country_flag = GWRMP_SI_OIL
								has_country_flag = GWRMP_SI_ALUMINUM
								has_country_flag = GWRMP_SI_RUBBER
								has_country_flag = GWRMP_SI_TUNGSTEN
								has_country_flag = GWRMP_SI_STEEL
								has_country_flag = GWRMP_SI_CHROMIUM
								has_country_flag = GWRMP_SI_COAL
							} 
							has_political_power > var:global.si_ask_resources_pp_required
							NOT = {	has_idea = SI_else_demands_idea	}
						}
					}
				}
			}

			state_transfer_mark_button_visible = {
				NOT = {
					ROOT = { has_country_flag = gwrmp_si_select_resources_visible }
					is_in_array = {
						array = ROOT.selected_states_array
						value = FROM.id
					}
				}
			}
			state_transfer_unmark_button_visible = {
				NOT = { ROOT = { has_country_flag = gwrmp_si_select_resources_visible } }
				is_in_array = {
					array = ROOT.selected_states_array
					value = FROM.id
				}
			}
			state_transfer_mark_button_click_enabled = {
				OR = {
					FROM = { is_owned_and_controlled_by = ROOT is_capital = no }
					FROM.owner = { 
						is_subject_of = ROOT 
						OR = {
							AND = { is_ai = no OR = { PREV = { is_capital = no is_ai = no} is_ai = yes} }
							is_ai = yes
						}
					}
				}
			}
			state_transfer_unmark_all_button_click_enabled = {
				any_state = {
					is_in_array = {
						array = ROOT.selected_states_array
						value = THIS.id
					}
				}
			}
			state_transfer_transfer_button_click_enabled = {
				any_state = {
					is_in_array = {
						array = ROOT.selected_states_array
						value = THIS.id
					}
				}
				FROM.OWNER = {
					NOT = { tag = GWR }
				}
			}
		}
	}
